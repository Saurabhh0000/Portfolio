// Fixly: Core booking logic with role-based access
@Service
@Transactional
public class BookingService {

    private final BookingRepository bookingRepo;
    private final OtpService         otpService;
    private final NotificationService notifier;

    @PreAuthorize("hasRole('CUSTOMER')")
    public BookingResponse createBooking(
            BookingRequest req,
            @AuthenticationPrincipal UserDetails user) {

        Booking booking = Booking.builder()
            .customer(user.getUsername())
            .providerId(req.getProviderId())
            .scheduledAt(req.getScheduledAt())
            .status(BookingStatus.PENDING)
            .build();

        bookingRepo.save(booking);
        otpService.sendVerificationOtp(booking);
        notifier.alertProvider(booking.getProviderId());

        return BookingMapper.toResponse(booking);
    }

    @PreAuthorize("hasRole('PROVIDER')")
    public void verifyJobOtp(Long bookingId, String otp) {
        otpService.validate(bookingId, otp);
        bookingRepo.updateStatus(bookingId, BookingStatus.IN_PROGRESS);
    }
}
